# ============ GEO 同步服务器（每次粘贴这段）============

# 1. 拉代码（先丢弃 package-lock 本地改动，防冲突）
cd /opt/geo
git checkout -- frontend/package-lock.json 2>/dev/null
git pull

# 2. 前端构建（限内存，低配机不卡）
cd /opt/geo/frontend
NODE_OPTIONS="--max-old-space-size=512" npm run build

# 3. 后端重启（端口 8088）
pkill -f uvicorn
cd /opt/geo/backend
nohup ./venv/bin/uvicorn app.main:app --host 0.0.0.0 --port 8088 > /tmp/geo-backend.log 2>&1 &
sleep 3
tail -n 20 /tmp/geo-backend.log
